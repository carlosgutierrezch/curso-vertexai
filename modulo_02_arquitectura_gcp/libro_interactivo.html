Paso 2: Autenticación con Service Account

Este patrón te da un cliente autenticado para BigQuery y Cloud Storage. Es el primer bloque que ejecutarás en cualquier script.

🐍 Python — auth.py
from google.cloud import bigquery, storage
from google.oauth2 import service_account
import os

def get_credentials():
    """Carga credenciales de service account desde disco o variable de entorno."""
    creds_path = os.environ.get(
        "GOOGLE_APPLICATION_CREDENTIALS",
        os.path.join(os.path.dirname(__file__), "service-account.json"),
    )
    return service_account.Credentials.from_service_account_file(creds_path)

def get_bq_client():
    """Devuelve un cliente autenticado de BigQuery."""
    return bigquery.Client(
        credentials=get_credentials(),
        project=PROJECT
    )

def get_gcs_client():
    """Devuelve un cliente autenticado de Cloud Storage."""
    return storage.Client(
        credentials=get_credentials(),
        project=PROJECT
    )
🔌
En Vertex AI Workbench no necesitas el archivo JSON. El notebook hereda las credenciales del entorno automáticamente. Puedes simplificar a: client = bigquery.Client(project=PROJECT)

Crear el dataset

El dataset es el contenedor de tus tablas. Lo creas una vez y es donde aplicas permisos a nivel de dominio de datos.

🐍 Python — crear dataset
from google.cloud import bigquery

def ensure_dataset(client, project_id, dataset_name):
    """Crea el dataset si no existe. Idempotente."""
    dataset_ref = f"{project_id}.{dataset_name}"
    try:
        client.get_dataset(dataset_ref)
        print(f"Dataset {dataset_ref} ya existe")
    except Exception:
        dataset = bigquery.Dataset(dataset_ref)
        dataset.location = "EU"  # Datos en Europa (GDPR)
        dataset.description = "Datos de People Analytics — Confidencial"
        client.create_dataset(dataset)
        print(f"Dataset {dataset_ref} creado")

# Uso:
client = get_bq_client()
ensure_dataset(client, PROJECT, DATASET)
🌎
Ubicación "EU": Con datos de empleados europeos, el dataset DEBE estar en región EU. BigQuery no permite mover datos entre regiones después de crearlos. Si tu empresa está en España, usa europe-west1 (Madrid) o EU (multi-región Europa).

Crear tabla con particionado temporal

El particionado es crucial: reduce costes (solo escaneas las fechas que necesitas) y mejora rendimiento.

🐍 Python — crear tabla particionada
def ensure_table(client, project_id, dataset, table_name, schema):
    """Crea tabla particionada si no existe."""
    table_ref = f"{project_id}.{dataset}.{table_name}"
    try:
        client.get_table(table_ref)
        print(f"Tabla {table_ref} ya existe")
    except Exception:
        table = bigquery.Table(table_ref, schema=schema)

        # Particionar por fecha — cada mes es una partición
        table.time_partitioning = bigquery.TimePartitioning(
            type_=bigquery.TimePartitioningType.DAY,
            field="fecha_snapshot",
        )
        # IMPORTANTE: forzar filtro de partición en queries
        # Evita escaneos completos accidentales (coste + GDPR)
        table.require_partition_filter = True

        client.create_table(table)
        print(f"Tabla {table_ref} creada con particionado por fecha_snapshot")

# Uso:
ensure_table(client, PROJECT, DATASET, "empleados", get_empleados_schema())
💰
require_partition_filter = True — Esta línea es tu mejor amiga. Sin ella, alguien puede hacer SELECT * FROM empleados y escanear TODOS los datos de TODOS los meses. Con ella, BigQuery rechaza queries sin filtro de fecha. Esto protege tu factura Y cumple con la minimización de datos de GDPR.

Leer archivos desde GCS

Este patrón lee todos los JSONs de una fecha específica. Lo usarás para ingestar datos que otros sistemas depositan en el bucket.

🐍 Python — leer blobs de GCS
import json
import logging

def read_files_from_gcs(storage_client, bucket_name, date):
    """Lee todos los archivos JSON de una fecha desde GCS."""
    prefix = f"rrhh/raw/sap/{date}/"
    bucket = storage_client.bucket(bucket_name)
    blobs = list(bucket.list_blobs(prefix=prefix))

    logging.info(f"Encontrados {len(blobs)} archivos para fecha {date}")

    registros = []
    errores = 0

    for blob in blobs:
        try:
            contenido = blob.download_as_text(encoding="utf-8")
            data = json.loads(contenido)
            registros.append((blob.name, data))
        except Exception as e:
            errores += 1
            logging.warning(f"Error leyendo {blob.name}: {e}")

    if errores:
        logging.warning(f"{errores} archivos no se pudieron leer")

    return registros

Escribir resultados a GCS

🐍 Python — escribir a GCS
import json
from datetime import datetime

def save_to_gcs(storage_client, bucket_name, data, filename):
    """Guarda un dict como JSON en GCS."""
    bucket = storage_client.bucket(bucket_name)
    today = datetime.now().strftime("%Y-%m-%d")
    blob_path = f"rrhh/processed/{today}/{filename}"

    blob = bucket.blob(blob_path)
    blob.upload_from_string(
        json.dumps(data, ensure_ascii=False, indent=2),
        content_type="application/json"
    )
    logging.info(f"Guardado: gs://{bucket_name}/{blob_path}")

Manejo de encodings

En RRHH los datos vienen de todas partes — SAP exporta en un encoding, las encuestas en otro. Este patrón intenta múltiples encodings:

🐍 Python — encoding robusto
def read_file_safe(filepath):
    """Lee un archivo intentando múltiples encodings."""
    for enc in ["utf-8", "utf-16", "latin-1", "cp1252"]:
        try:
            with open(filepath, "r", encoding=enc) as f:
                return f.read()
        except (UnicodeDecodeError, UnicodeError):
            continue
    raise ValueError(f"No se pudo decodificar: {filepath}")

import pandas as pd
from google.cloud import bigquery

def upload_dataframe(client, df, table_ref, mode="WRITE_APPEND"):
    """Sube un DataFrame a BigQuery.

    Args:
        client: BigQuery client autenticado
        df: DataFrame de pandas
        table_ref: 'proyecto.dataset.tabla'
        mode: WRITE_APPEND (añadir) o WRITE_TRUNCATE (reemplazar)
    """
    job_config = bigquery.LoadJobConfig(
        write_disposition=mode
    )
    job = client.load_table_from_dataframe(df, table_ref, job_config=job_config)
    job.result()  # Esperar a que termine

    logging.info(f"Subidas {len(df)} filas a {table_ref}")

def insert_rows_json(client, table_ref, rows, schema):
    """Inserta filas JSON en BigQuery con validación de schema."""
    if not rows:
        logging.info("No hay filas para insertar")
        return 0

    job_config = bigquery.LoadJobConfig(
        schema=schema,
        write_disposition=bigquery.WriteDisposition.WRITE_APPEND,
    )

    load_job = client.load_table_from_json(
        rows, table_ref, job_config=job_config
    )
    load_job.result()

    logging.info(f"Insertadas {len(rows)} filas en {table_ref}")
    return len(rows)

Transformar datos antes de subir

Patron para extraer campos de un JSON crudo a un dict listo para BigQuery:

🐍 Python — transformar row
from typing import Any

def safe_get(mapping: dict, *keys: str, default: Any = None) -> Any:
    """Acceso seguro a dicts anidados sin KeyError."""
    current = mapping
    for key in keys:
        if isinstance(current, dict):
            current = current.get(key)
        else:
            return default
        if current is None:
            return default
    return current

def transform_sap_record(raw_data):
    """Transforma un registro crudo de SAP a formato BigQuery."""
    return {
        "empleado_id":        safe_get(raw_data, "PersonalInfo", "employeeId"),
        "departamento":       safe_get(raw_data, "JobInfo", "department"),
        "nivel":              safe_get(raw_data, "JobInfo", "jobLevel"),
        "salario_bruto":      safe_get(raw_data, "CompInfo", "annualSalary", default=0),
        "fecha_alta":         safe_get(raw_data, "EmploymentInfo", "hireDate"),
        "rating_desempeno":   safe_get(raw_data, "Performance", "overallRating"),
    }

SQL para People Analytics

Con los datos en BigQuery, ahora viene la parte más poderosa: hacer preguntas a los datos. Estas son queries reales que responden preguntas que RRHH hace todos los días.

Query 1: Dashboard de rotación por departamento

📊 SQL — rotación por departamento
-- Tasa de rotación por departamento en el último año
SELECT
    departamento,
    COUNT(*) AS total_empleados,
    COUNTIF(rotacion = TRUE) AS bajas,
    ROUND(COUNTIF(rotacion = TRUE) * 100.0 / COUNT(*), 1) AS tasa_rotacion_pct,
    ROUND(AVG(salario_bruto), 0) AS salario_medio,
    ROUND(AVG(score_clima), 1) AS clima_medio
FROM `people-analytics-prod.people_analytics.empleados`
WHERE fecha_snapshot = '2026-03-01'  -- Filtro de partición obligatorio
GROUP BY departamento
ORDER BY tasa_rotacion_pct DESC

Query 2: Brecha salarial de género controlada por nivel

📊 SQL — brecha salarial
-- Brecha salarial ajustada por nivel jerárquico
WITH salarios AS (
    SELECT
        nivel,
        genero,
        AVG(salario_bruto) AS salario_medio,
        COUNT(*) AS n
    FROM `people-analytics-prod.people_analytics.empleados`
    WHERE fecha_snapshot = '2026-03-01'
      AND genero IN ('Hombre', 'Mujer')
    GROUP BY nivel, genero
)
SELECT
    h.nivel,
    ROUND(h.salario_medio, 0) AS salario_hombres,
    ROUND(m.salario_medio, 0) AS salario_mujeres,
    ROUND((h.salario_medio - m.salario_medio) / h.salario_medio * 100, 1) AS brecha_pct,
    h.n AS n_hombres,
    m.n AS n_mujeres
FROM salarios h
JOIN salarios m ON h.nivel = m.nivel
WHERE h.genero = 'Hombre' AND m.genero = 'Mujer'
ORDER BY brecha_pct DESC

Query 3: Detección de patrones de absentismo con REGEXP

📊 SQL — patrones de absentismo
-- Empleados con patrón de absentismo preocupante
-- (alto absentismo + bajo clima + mucho tiempo sin promoción)
SELECT
    empleado_id,
    departamento,
    nivel,
    dias_absentismo,
    score_clima,
    meses_sin_promocion,
    rating_desempeno,
    -- Score de riesgo compuesto
    ROUND(
        (dias_absentismo / 60.0) * 0.4 +
        ((10 - score_clima) / 10.0) * 0.3 +
        (LEAST(meses_sin_promocion, 60) / 60.0) * 0.3
    , 3) AS risk_score
FROM `people-analytics-prod.people_analytics.empleados`
WHERE fecha_snapshot = '2026-03-01'
  AND rotacion = FALSE  -- Solo empleados activos
  AND dias_absentismo > 15
ORDER BY risk_score DESC
LIMIT 50

def query_rotacion_por_depto(client, fecha_snapshot):
    """Ejecuta query de rotación y devuelve DataFrame."""
    sql = f"""
    SELECT
        departamento,
        COUNT(*) AS total,
        COUNTIF(rotacion = TRUE) AS bajas,
        ROUND(COUNTIF(rotacion = TRUE) * 100.0 / COUNT(*), 1) AS tasa_pct
    FROM `{PROJECT}.{DATASET}.empleados`
    WHERE fecha_snapshot = '{fecha_snapshot}'
    GROUP BY departamento
    ORDER BY tasa_pct DESC
    """
    return client.query(sql).to_dataframe()

ETL: Bronze → Silver → Gold

Este es el corazón de tu infraestructura de datos. La arquitectura medallion organiza los datos en tres capas de calidad creciente. Es el patrón más usado en proyectos de datos modernos.

   BRONZE (raw)             SILVER (clean)            GOLD (business)
┌─────────────────┐    ┌─────────────────┐    ┌─────────────────────┐
│ Datos crudos    │    │ Datos limpios   │    │ Métricas de negocio │
│ tal cual llegan │───►│ validados       │───►│ listas para consumo │
│                 │    │ normalizados    │    │                     │
│ SAP exports     │    │ Sin duplicados  │    │ KPIs de rotación    │

import time
import logging
import pandas as pd

def run_pipeline(client, fecha, steps="all"):
    """Ejecuta el pipeline ETL completo Bronze → Silver → Gold."""
    log.info(f"{'═' * 60}")
    log.info(f"ETL PEOPLE ANALYTICS — fecha={fecha}")
    log.info(f"{'═' * 60}")

    t_total = time.time()
    df_bronze = None
    df_silver = None

    # BRONZE: Ingestar datos crudos
    if steps in ("all", "bronze"):
        df_bronze = run_bronze(client, fecha)

    # SILVER: Limpiar y normalizar
    if steps in ("all", "silver"):
        df_silver = run_silver(client, fecha, df_bronze)

    # GOLD: Calcular métricas de negocio
    if steps in ("all", "gold"):
        run_gold(client, fecha, df_silver)

    log.info(f"{'═' * 60}")
    log.info(f"ETL COMPLETADO en {time.time() - t_total:.0f}s")
    log.info(f"{'═' * 60}")

def run_bronze(client, fecha):
    """BRONZE: Ingesta de datos crudos desde GCS a BigQuery."""
    log.info("[BRONZE] Iniciando ingesta...")
    t = time.time()

    table_ref = f"{PROJECT}.{DATASET}.bronze_empleados"

    # PASO 1: Borrar datos existentes de esta fecha (idempotencia)
    delete_partition(client, table_ref, fecha)

    # PASO 2: Leer datos crudos desde GCS
    gcs_client = get_gcs_client()
    registros = read_files_from_gcs(gcs_client, GCS_BUCKET, fecha)

    # PASO 3: Transformar a filas de BigQuery
    rows = []
    for blob_name, data in registros:
        row = transform_sap_record(data)
        row["fecha_snapshot"] = fecha
        row["source_file"] = blob_name
        rows.append(row)

    # PASO 4: Insertar en BigQuery
    insert_rows_json(client, table_ref, rows, get_empleados_schema())

    df = pd.DataFrame(rows)
    log.info(f"[BRONZE] {len(rows)} registros en {time.time()-t:.1f}s")
    return df

DELETE antes de INSERT: la clave de la idempotencia

Este es el patrón más importante del ETL. Permite re-ejecutar el pipeline sin miedo a duplicados:

🐍 Python — delete idempotente
def delete_partition(client, table_ref, fecha):
    """Borra filas de una fecha para permitir re-ejecución segura."""
    query = f"DELETE FROM `{table_ref}` WHERE fecha_snapshot = '{fecha}'"
    try:
        client.query(query).result()
        log.info(f"Borradas filas existentes para {fecha}")
    except Exception as e:
        log.debug(f"Sin filas previas para {fecha}: {e}")
🔄
Idempotencia significa que puedes ejecutar el mismo pipeline 10 veces y el resultado es el mismo que ejecutarlo 1 vez. Es crítico en producción: si algo falla a mitad, simplemente re-ejecutas todo sin preocuparte por duplicados.

Capas SILVER y GOLD

🐍 Python — capas Silver y Gold
def run_silver(client, fecha, df_bronze=None):
    """SILVER: Limpieza, normalización y enriquecimiento."""
    log.info("[SILVER] Limpiando y normalizando...")

    # Si no tenemos el DataFrame, leerlo de la tabla bronze
    if df_bronze is None:
        sql = f"""SELECT * FROM `{PROJECT}.{DATASET}.bronze_empleados`
                  WHERE fecha_snapshot = '{fecha}'"""
        df_bronze = client.query(sql).to_dataframe()

    df = df_bronze.copy()

    # Normalizar departamentos (variaciones de nombre)
    df["departamento"] = df["departamento"].str.strip().str.title()

    # Eliminar duplicados por empleado_id (quedarse con el último)
    df = df.drop_duplicates(subset=["empleado_id"], keep="last")

    # Validar rangos
    df["salario_bruto"] = df["salario_bruto"].clip(15000, 200000)
    df["score_clima"] = df["score_clima"].clip(1, 10)
    df["rating_desempeno"] = df["rating_desempeno"].clip(1, 5)

    # Subir a tabla silver
    table_ref = f"{PROJECT}.{DATASET}.silver_empleados"
    delete_partition(client, table_ref, fecha)
    upload_dataframe(client, df, table_ref)

    log.info(f"[SILVER] {len(df)} registros limpios")
    return df


def run_gold(client, fecha, df_silver=None):
    """GOLD: Calcular métricas de negocio."""
    log.info("[GOLD] Calculando métricas de negocio...")

    # Métricas calculadas directamente en SQL (más eficiente)
    sql = f"""
    INSERT INTO `{PROJECT}.{DATASET}.gold_metricas_depto`
    SELECT
        fecha_snapshot,
        departamento,
        COUNT(*) AS headcount,
        COUNTIF(rotacion = TRUE) AS bajas,
        ROUND(AVG(salario_bruto), 0) AS salario_medio,
        ROUND(AVG(score_clima), 1) AS clima_medio,
        ROUND(AVG(dias_absentismo), 1) AS absentismo_medio,
        ROUND(AVG(rating_desempeno), 2) AS rating_medio,
        ROUND(AVG(horas_formacion), 1) AS formacion_media
    FROM `{PROJECT}.{DATASET}.silver_empleados`
    WHERE fecha_snapshot = '{fecha}'
    GROUP BY fecha_snapshot, departamento
    """

    # Borrar métricas existentes primero (idempotencia)
    table_ref = f"{PROJECT}.{DATASET}.gold_metricas_depto"
    delete_partition(client, table_ref, fecha)
    client.query(sql).result()

    log.info("[GOLD] Métricas por departamento calculadas")

Tablas externas y vistas

A veces no quieres copiar datos a BigQuery — quieres consultarlos directamente donde están (en GCS). Las tablas externas hacen exactamente eso.

Crear tabla externa sobre archivos en GCS

Esto es muy útil para logs, exports temporales, o datos que cambian frecuentemente:

🐍 Python — tabla externa GCS → BigQuery
def create_external_table(client, project, dataset):
    """Crea tabla externa que lee JSONs directamente desde GCS."""
    table_ref = f"{project}.{dataset}.raw_encuestas_ext"
    table = bigquery.Table(table_ref)

    # Cada línea JSON es una fila con un campo "data" tipo STRING
    external_config = bigquery.ExternalConfig("CSV")
    external_config.source_uris = [
        "gs://people-analytics-datalake/rrhh/raw/encuestas/*"
    ]
    external_config.autodetect = False
    external_config.schema = [
        bigquery.SchemaField("data", "STRING")
    ]

    # Truco: usar un delimitador que nunca aparece en JSON
    csv_opts = bigquery.CSVOptions()
    csv_opts.field_delimiter = "\x01"          # Carácter SOH
    csv_opts.allow_quoted_newlines = True
    csv_opts.quote_character = ""
    external_config.csv_options = csv_opts

    table.external_data_configuration = external_config

    # Recrear (delete + create para idempotencia)
    client.delete_table(table_ref, not_found_ok=True)
    client.create_table(table)
    log.info(f"Tabla externa creada: {table_ref}")

Consultar tabla externa con JSON_VALUE

📊 SQL — consultar tabla externa
-- Extraer campos de JSONs en GCS sin copiarlos a BigQuery
SELECT
    JSON_VALUE(data, '$.empleado_id')           AS empleado_id,
    CAST(JSON_VALUE(data, '$.fecha') AS DATE)  AS fecha,
    CAST(JSON_VALUE(data, '$.score') AS FLOAT64) AS score_clima,

FROM `people-analytics-prod.people_analytics.raw_encuestas_ext`
WHERE JSON_VALUE(data, '$.departamento') = 'Tecnología'
🔌
Cuándo usar tablas externas vs. tablas nativas:
Externas: Exploración, datos temporales, logs que cambian constantemente.
Nativas: Datos que consultas frecuentemente — son más rápidas y baratas por query.

Vistas para simplificar queries complejas

📊 SQL — vista de riesgo de rotación
-- Vista que RRHH puede consultar sin saber SQL complejo
CREATE OR REPLACE VIEW `people_analytics.v_rotacion_risk` AS
SELECT
    empleado_id,
    departamento,
    nivel,
    antiguedad_meses,
    meses_sin_promocion,
    score_clima,
    dias_absentismo,
    ROUND(
        (LEAST(meses_sin_promocion, 60) / 60.0) * 0.30 +
        ((10 - score_clima) / 10.0) * 0.25 +
        (dias_absentismo / 60.0) * 0.15 +
        (CASE WHEN rating_desempeno <= 2 THEN 0.3 ELSE 0.0 END)
    , 3) AS risk_score,
    CASE
        WHEN risk_score > 0.6 THEN 'ALTO'
        WHEN risk_score > 0.35 THEN 'MEDIO'
        ELSE 'BAJO'
    END AS nivel_riesgo,
    fecha_snapshot
FROM `people_analytics.silver_empleados`
WHERE rotacion = FALSE

def parse_json_safe(data):
    """Parsea JSON con fallback para datos legacy."""
    if isinstance(data, str):
        try:
            return json.loads(data)
        except Exception:
            return {}
    return data if isinstance(data, dict) else {}

def safe_get(mapping, *keys, default=None):
    """Acceso seguro a dicts anidados."""
    current = mapping
    for key in keys:
        if isinstance(current, dict):
            current = current.get(key)
        else:
            return default
        if current is None:
            return default
    return current

# Ejemplo real: datos de SAP vienen con estructura variable
